        // CONTRASEÑA POR DEFECTO CIFRADA (Para cambiarla, puedes modificar este script)

        // La contraseña por defecto instalada aquí es: "admin123"

        const SECRET_HASH = "8c6976e5b5410415bde908bd4dee15dfb167a9c873fc4bb8a81f6f2ab448a918";

        // Algoritmo de encriptación SHA-256 para validación segura en cliente

        async function sha256(message) {

            const msgBuffer = new TextEncoder().encode(message);                    

            const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);

            const hashArray = Array.from(new Uint8Array(hashBuffer));

            return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

        }

        window.onload = function() {

        function loadWebsiteData() {

            const fields = ['hero-title', 'hero-desc', 'card1-title', 'card1-desc', 'card2-title', 'card2-desc', 'card3-title', 'card3-desc'];

            fields.forEach(field => {

                const savedValue = localStorage.getItem('btcminers-' + field);

                if(savedValue) {

                    document.getElementById('view-' + field).innerHTML = savedValue;

                }

            });

        }

        function openAdminFlow() {

            if (sessionStorage.getItem('isBtcAdmin') === 'true') {

                showAdminPanel();

            } else {

                document.getElementById('loginModal').style.display = 'flex';

            }

        }

        function closeLoginModal() {

            document.getElementById('loginModal').style.display = 'none';

            document.getElementById('adminPassword').value = '';

        }

        async function verifyAdminPassword() {

            const inputPass = document.getElementById('adminPassword').value;

            const hashedInput = await sha256(inputPass);

            if(hashedInput === SECRET_HASH) {

                sessionStorage.setItem('isBtcAdmin', 'true');

                closeLoginModal();

                showAdminPanel();

            } else {

                alert('Acceso Denegado: Credencial incorrecta.');

                closeLoginModal();

            }

        }

        function showAdminPanel() {

            document.getElementById('adminPanel').style.display = 'block';

           

            // Rellenar campos del panel con lo que hay actualmente en pantalla

            });

        }

        function closeAdminPanel() {

            document.getElementById('adminPanel').style.display = 'none';

        }

        function saveWebChanges() {

            if (sessionStorage.getItem('isBtcAdmin') !== 'true') return;

            const fields = ['hero-title', 'hero-desc', 'card1-title', 'card1-desc', 'card2-title', 'card2-desc', 'card3-title', 'card3-desc'];

           

            fields.forEach(field => {

                const newValue = document.getElementById('input-' + field).value;

                localStorage.setItem('btcminers-' + field, newValue);

            });

            alert('¡Cambios guardados con éxito de manera segura!');

            closeAdminPanel();

        }
